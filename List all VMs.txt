// List all VMs
where type == 'microsoft.compute/virtualmachines' 
| project 	Computername=name, 
			ResourceGroup=resourceGroup,
			VMSize=properties.hardwareProfile.vmSize,
			Publisher=properties.storageProfile.imageReference.publisher, 
			Version=properties.storageProfile.imageReference.version, 
			Offer=properties.storageProfile.imageReference.offer, 
			Sku=properties.storageProfile.imageReference.sku,
			ExactVersion=properties.storageProfile.imageReference.exactVersion,
			// DiskCaching=properties.storageProfile.osDisk.caching,
			OSType=properties.storageProfile.osDisk.osType,
			// OSDiskEncryption=properties.storageProfile.osDisk.encryptionSettings.enabled,
			// BootDiagnostics=properties.diagnosticsProfile.bootDiagnostics.enabled,
			Region=location,
			SubscriptionId=subscriptionId,
			State=properties.extended.instanceView.powerState.displayStatus,
			AvailabilityZone = tostring(zones[0]),
			tag_Environment=tags.Environment,
			tag_Owner=tags.Owner,
			tag_Techops=tags.Techops,
			tag_ProductFamily=tags.["Product Family"],
			tag_Group=tags.Group,
            tag_Function=tags.Function,
            tag_Role=tags.Role,
            tag_Shutdown=tags.Shutdown,
            tag_Client = tags.Client
