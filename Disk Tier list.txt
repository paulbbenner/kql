
// Click the "Run query" command above to execute the query and see results.
resources
| where type =~ "microsoft.compute/disks"
| where name !contains 'osdisk' 
| extend diskState = tostring(properties.diskState)
     , storageType = tostring(sku.name)
     , diskTier = properties.diskSku.tier
     , diskSizeGB = todouble(properties.diskSizeGB)
     , osType = tostring(properties.osType)
     , look = tostring(properties.purchasePlan.name)
| extend pathParts = parse_path(managedBy)
| extend OwnerVM = tostring(pathParts.Filename)
| extend TagsObject = parse_json(tags)
| extend OwnerTag = tostring(TagsObject.Owner)
     , OwnerEmailTag = tostring(TagsObject.['Owner Email'])
     , CreatedByTag = tostring(TagsObject.['Created By'])
     , ProductFamilyTag = tostring(TagsObject.['Product Family'])
     , SystemTag = tostring(TagsObject.System)
     , SubSystemTag = tostring(TagsObject.Subsystem)
| extend diskTier = case( storageType =~ 'Standard_LRS' and diskSizeGB < 8, 'E1',
                         storageType =~ 'Standard_LRS' and diskSizeGB < 16, 'E2',
                         storageType =~ 'Standard_LRS' and diskSizeGB < 32, 'E3',
                         storageType =~ 'Standard_LRS' and diskSizeGB < 64, 'E4',
                         storageType =~ 'Standard_LRS' and diskSizeGB < 128, 'E6',
                         storageType =~ 'Standard_LRS' and diskSizeGB < 256, 'E10',
                         storageType =~ 'Standard_LRS' and diskSizeGB < 512, 'E15',
                         storageType =~ 'Standard_LRS' and diskSizeGB < 1024, 'E20',
                         storageType =~ 'Standard_LRS' and diskSizeGB < 2048, 'E30',
                         storageType =~ 'Standard_LRS' and diskSizeGB < 4096, 'E40',
                         storageType =~ 'Standard_LRS' and diskSizeGB < 8192, 'E50',
                         storageType =~ 'Standard_LRS' and diskSizeGB < 16384, 'E60',
                         storageType =~ 'Standard_LRS' and diskSizeGB < 32768, 'E70',
                         storageType =~ 'Standard_LRS', 'E80',
                         storageType =~ 'Premium_LRS' and diskSizeGB < 8, 'P1',
                         storageType =~ 'Premium_LRS' and diskSizeGB < 16, 'P2',
                         storageType =~ 'Premium_LRS' and diskSizeGB < 32, 'P3',
                         storageType =~ 'Premium_LRS' and diskSizeGB < 64, 'P4',
                         storageType =~ 'Premium_LRS' and diskSizeGB < 128, 'P6',
                         storageType =~ 'Premium_LRS' and diskSizeGB < 256, 'P10',
                         storageType =~ 'Premium_LRS' and diskSizeGB < 512, 'P15',
                         storageType =~ 'Premium_LRS' and diskSizeGB < 1024, 'P20',
                         storageType =~ 'Premium_LRS' and diskSizeGB < 2048, 'P30',
                         storageType =~ 'Premium_LRS' and diskSizeGB < 4096, 'P40',
                         storageType =~ 'Premium_LRS' and diskSizeGB < 8192, 'P50',
                         storageType =~ 'Premium_LRS' and diskSizeGB < 16384, 'P60',
                         storageType =~ 'Premium_LRS' and diskSizeGB < 32768, 'P70',
                         storageType =~ 'Premium_LRS', 'P80',
                         '')
| join kind=inner (
    resourcecontainers
    | where type =~ 'microsoft.resources/subscriptions'
    | project subscriptionId, subscriptionName = name)
    on subscriptionId
| project subscriptionName
     , region = location
     , diskState
     , storageType
     , diskTier
     , look
     , diskSizeGB
     , resourceGroup
     , OwnerVM
     , osType
     , name
     , OwnerTag
     , OwnerEmailTag
     , CreatedByTag
     , ProductFamilyTag
     , SystemTag
     , SubSystemTag 
     , ['tags']
//| where OwnerVM == ""  // unattached disks
| order by subscriptionName asc
     , region asc
     , diskState asc
     , storageType asc
     , diskTier asc
     , diskSizeGB asc
     , resourceGroup asc
     , OwnerVM asc
     , name asc
